JavaScript: Add Condition Row
function addConditionRow(condition = {}) {
  const row = document.createElement('div');
  row.className = 'condition-row';
  
  // Question selector
  const questionSelect = document.createElement('select');
  questionSelect.className = 'form-control';
  schema.categories.forEach(cat => {
    cat.questions.forEach(q => {
      if (q.id !== currentQuestionId) {
        const opt = new Option(
          `${cat.label}: ${q.label} (${q.id})`,
          q.id
        );
        questionSelect.add(opt);
      }
    });
  });
  questionSelect.value = condition.questionId || '';
  
  // Operator selector
  const operatorSelect = document.createElement('select');
  operatorSelect.className = 'form-control';
  ['equals', 'notEquals', 'contains', 'notContains'].forEach(op => {
    operatorSelect.add(new Option(op, op));
  });
  operatorSelect.value = condition.operator || 'equals';
  
  // Value input (populate based on question type)
  const valueInput = createValueInput(condition);
  
  // Remove button
  const removeBtn = document.createElement('button');
  removeBtn.className = 'btn btn-danger btn-sm';
  removeBtn.textContent = '×';
  removeBtn.onclick = () => row.remove();
  
  row.append(questionSelect, operatorSelect, valueInput, removeBtn);
  document.getElementById('conditionsContainer').appendChild(row);
}
JavaScript: Save Conditions
function saveVisibleWhen() {
  const conditions = [];
  
  document.querySelectorAll('.condition-row').forEach(row => {
    const inputs = row.querySelectorAll('select, input');
    conditions.push({
      questionId: inputs[0].value,
      operator: inputs[1].value,
      value: inputs[2].value
    });
  });
  
  // Update question object
  currentQuestion.visibleWhen = conditions.length > 0 
    ? conditions 
    : undefined;
  
  // Save to backend
  saveSchema();
}
JavaScript: Evaluate Visibility
function isQuestionVisible(question, answers) {
  if (!question.visibleWhen || question.visibleWhen.length === 0) {
    return true; // No conditions = always visible
  }
  
  // ALL conditions must be true (AND logic)
  return question.visibleWhen.every(condition => {
    const answerValue = answers[condition.questionId];
    
    switch (condition.operator) {
      case 'equals':
        return answerValue === condition.value;
      
      case 'notEquals':
        return answerValue !== condition.value;
      
      case 'contains':
        // For multiselect (array values)
        return Array.isArray(answerValue) 
          ? answerValue.includes(condition.value)
          : answerValue === condition.value;
      
      case 'notContains':
        return Array.isArray(answerValue)
          ? !answerValue.includes(condition.value)
          : answerValue !== condition.value;
      
      default:
        return false;
    }
  });
}
Supported Operators Reference
Complete list of conditional operators with use cases
equals

Checks if the answer exactly matches the specified value. Use for radio buttons and dropdowns. 

{
  "operator": "equals",
  "value": "Azure"
}
notEquals

Checks if the answer is different from the specified value. Use for exclusion logic. 

{
  "operator": "notEquals",
  "value": "None"
}
contains

For multiselect answers, checks if the array includes the value. Most common for multiselect questions. 

{
  "operator": "contains",
  "value": "Mobile application"
}
notContains

For multiselect answers, checks if the array does NOT include the value. 

{
  "operator": "notContains",
  "value": "Legacy system"
}
